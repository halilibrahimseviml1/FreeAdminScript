        Position = UDim2.new(1, -190, 0.5, -20)
    }):Play()

    task.delay(2, function()
        local tweenOut = TweenService:Create(NotifFrame, TweenInfo.new(0.3, Enum.EasingStyle.Quad, Enum.EasingDirection.In), {
            Position = UDim2.new(1, 10, 0.5, -20)
        })
        tweenOut:Play()
        tweenOut.Completed:Connect(function()
            NotifFrame:Destroy()
        end)
    end)
end

-- Update Commands Logic
local function updateCommands()
    local prefix = PrefixBox.Text
    local inputCmd = TextBox.Text:gsub("^%s+", ""):gsub("%s+$", "")
    
    if inputCmd == "" then
        OutputLabel.Text = ""
        return
    end
    
    local resultText = ""
    for _, player in ipairs(Players:GetPlayers()) do
        if includeSelf or player ~= LocalPlayer then
            resultText = resultText .. prefix .. inputCmd .. " " .. player.Name .. "\n"
        end
    end
    OutputLabel.Text = resultText
end

-- Toggle Logic
ToggleButton.MouseButton1Click:Connect(function()
    includeSelf = not includeSelf
    if includeSelf then
        ToggleButton.Text = "Include Self: ON"
        ToggleButton.BackgroundColor3 = Color3.fromRGB(0, 180, 80)
    else
        ToggleButton.Text = "Include Self: OFF"
        ToggleButton.BackgroundColor3 = Color3.fromRGB(60, 60, 60)
    end
    updateCommands()
end)

PrefixBox:GetPropertyChangedSignal("Text"):Connect(updateCommands)
TextBox:GetPropertyChangedSignal("Text"):Connect(updateCommands)
Players.PlayerAdded:Connect(updateCommands)
Players.PlayerRemoving:Connect(updateCommands)

updateCommands()
